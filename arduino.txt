#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <WiFi.h>
#include <WiFiManager.h>   // "WiFiManager" by tzapu (v2.0.17 recommended)
#include <HTTPClient.h>
#include "DHT.h"

// ================= WIFI =================
// No SSID/password is stored in this source code.
// Credentials are entered through the WiFiManager portal and saved in the
// ESP32's flash (NVS), so they survive restarts and power loss.
#define AP_NAME "DETECT-CO-Setup"          // setup access point (open, portal at 192.168.4.1)

const unsigned long CONNECT_TIMEOUT_S          = 20;      // max wait when joining a saved network
const unsigned long PORTAL_TIMEOUT_S           = 180;     // portal auto-closes after 3 min
const unsigned long WIFI_RECONNECT_INTERVAL_MS = 15000;   // retry saved network every 15 s
const unsigned long WIFI_PORTAL_REOPEN_MS      = 300000;  // reopen portal after 5 min offline

// Config button: BOOT button on most ESP32 DevKit boards (GPIO0, otherwise unused here)
#define CONFIG_BUTTON_PIN 0
const unsigned long BUTTON_PORTAL_HOLD_MS = 3000;    // hold 3 s  -> open portal
const unsigned long BUTTON_ERASE_HOLD_MS  = 10000;   // hold 10 s -> erase saved WiFi + open portal

WiFiManager wm;
bool wifiWasConnected = false;
bool portalWasActive = false;
bool disconnectTracking = false;
unsigned long disconnectedSince = 0;
unsigned long lastReconnectAttempt = 0;

// ================= FIREBASE =================
const char* firebaseURL = "https://test1-thesisgrp20-default-rtdb.asia-southeast1.firebasedatabase.app/flood.json";

// ================= OLED =================
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET    -1
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

// ================= SENSORS =================
// Ultrasonic JSN-SR04T
#define TRIG_PIN 2
#define ECHO_PIN 5

// Water sensor
#define WATER_PIN 4

// KY-015 (DHT11) combi sensor
#define KY015_PIN 18
#define DHTTYPE DHT11
DHT ky015(KY015_PIN, DHTTYPE);

// Firebase update timing
unsigned long previousFirebase = 0;
const long FIREBASE_INTERVAL = 1000; // 1 second

// Sensor readings
float distance = 0;
bool waterDetected = false;
float temperature = 0;
float humidity = 0;

// ================= WIFI HELPERS =================
void onPortalStart(WiFiManager* w) {
  Serial.println("Config portal started.");
  Serial.println("Connect to WiFi: " AP_NAME);
  Serial.println("Then open http://192.168.4.1");
}

void onCredentialsSaved() {
  Serial.println("WiFi credentials saved.");
}

// Start the configuration portal without blocking the sensors.
void openConfigPortal() {
  if (wm.getConfigPortalActive()) return;
  Serial.println("Opening WiFi config portal...");
  wm.setConfigPortalTimeout(PORTAL_TIMEOUT_S);
  wm.startConfigPortal(AP_NAME);   // non-blocking: returns immediately
  portalWasActive = true;
}

void eraseWiFiCredentials() {
  Serial.println("Erasing saved WiFi credentials...");
  wm.resetSettings();
  wifiWasConnected = false;
  openConfigPortal();
}

// Must be called every loop(): runs the portal and handles reconnection.
void serviceWiFi() {
  wm.process();   // does nothing when the portal is not active

  unsigned long now = millis();
  bool portalActive = wm.getConfigPortalActive();

  // Portal just closed (saved successfully, or timed out)
  if (portalWasActive && !portalActive) {
    portalWasActive = false;
    WiFi.mode(WIFI_STA);
    if (WiFi.status() != WL_CONNECTED) {
      Serial.println("Portal closed without a connection. Will keep retrying.");
      disconnectTracking = true;
      disconnectedSince = now;
      lastReconnectAttempt = now;
      if (wm.getWiFiIsSaved()) WiFi.begin();
    }
  }

  if (portalActive) {
    portalWasActive = true;
    return;   // portal handles the connection while it is open
  }

  if (WiFi.status() == WL_CONNECTED) {
    if (!wifiWasConnected) {
      wifiWasConnected = true;
      Serial.print("WiFi connected. IP: ");
      Serial.println(WiFi.localIP());
    }
    disconnectTracking = false;
    return;
  }

  // ---- Not connected ----
  if (wifiWasConnected) {
    wifiWasConnected = false;
    Serial.println("WiFi connection lost.");
  }
  if (!disconnectTracking) {
    disconnectTracking = true;
    disconnectedSince = now;
    lastReconnectAttempt = now;
  }

  // Offline too long (bad/changed credentials or router gone): reopen portal
  if (now - disconnectedSince >= WIFI_PORTAL_REOPEN_MS) {
    disconnectedSince = now;
    openConfigPortal();
    return;
  }

  // Periodic reconnect attempt to the saved network
  if (now - lastReconnectAttempt >= WIFI_RECONNECT_INTERVAL_MS) {
    lastReconnectAttempt = now;
    if (wm.getWiFiIsSaved()) {
      Serial.println("Retrying saved WiFi network...");
      WiFi.disconnect();   // does not erase saved credentials
      WiFi.begin();
    }
  }
}

// Hold the BOOT button: 3 s = open portal, 10 s = erase credentials.
void handleConfigButton() {
  static unsigned long pressStart = 0;
  static bool portalTriggered = false;
  static bool eraseTriggered = false;

  if (digitalRead(CONFIG_BUTTON_PIN) == LOW) {
    if (pressStart == 0) pressStart = millis();
    unsigned long held = millis() - pressStart;
    if (!portalTriggered && held >= BUTTON_PORTAL_HOLD_MS) {
      portalTriggered = true;
      openConfigPortal();
    }
    if (!eraseTriggered && held >= BUTTON_ERASE_HOLD_MS) {
      eraseTriggered = true;
      eraseWiFiCredentials();
    }
  } else {
    pressStart = 0;
    portalTriggered = false;
    eraseTriggered = false;
  }
}

// Serial Monitor commands: 'p' = open portal, 'e' = erase saved WiFi
void handleSerialCommands() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == 'p') openConfigPortal();
    else if (c == 'e') eraseWiFiCredentials();
  }
}

void setup() {
  Serial.begin(115200);

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(WATER_PIN, INPUT);

  pinMode(CONFIG_BUTTON_PIN, INPUT_PULLUP);

  // Initialize I2C for OLED
  Wire.begin(21, 22);

  // Initialize OLED
  if(!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("SSD1306 allocation failed");
    for(;;);
  }

  display.clearDisplay();
  display.setTextSize(1);  // normal size
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0,0);
  display.println("Flood Monitor");
  display.println("Connecting WiFi...");
  display.display();

  // Initialize KY-015
  ky015.begin();

  // ================= WIFI MANAGER =================
  WiFi.mode(WIFI_STA);
  WiFi.setAutoReconnect(true);

  wm.setDebugOutput(false);                 // never print credentials to Serial
  wm.setConfigPortalBlocking(false);        // sensors keep running while portal is open
  wm.setConnectTimeout(CONNECT_TIMEOUT_S);
  wm.setConfigPortalTimeout(PORTAL_TIMEOUT_S);
  wm.setAPStaticIPConfig(IPAddress(192,168,4,1), IPAddress(192,168,4,1), IPAddress(255,255,255,0));
  wm.setAPCallback(onPortalStart);
  wm.setSaveConfigCallback(onCredentialsSaved);

  Serial.println("Starting WiFi Manager...");
  Serial.println("If no saved WiFi is found, connect to:");
  Serial.println(AP_NAME);

  // Tries saved credentials (max CONNECT_TIMEOUT_S). If that fails, starts the
  // portal in non-blocking mode and returns false. No restart, no endless loop.
  if (wm.autoConnect(AP_NAME)) {
    wifiWasConnected = true;
    Serial.println("\nWiFi Connected!");
    Serial.print("IP Address: ");
    Serial.println(WiFi.localIP());
  } else {
    portalWasActive = wm.getConfigPortalActive();
    Serial.println("WiFi not connected. Running offline; portal open if enabled.");
  }
}

void loop() {
  // ================= WIFI SERVICE =================
  serviceWiFi();
  handleConfigButton();
  handleSerialCommands();

  // ================= READ ULTRASONIC =================
  long duration;
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  duration = pulseIn(ECHO_PIN, HIGH, 50000);
  distance = (duration == 0) ? -1 : duration * 0.0343 / 2.0;

  // ================= READ WATER SENSOR =================
  waterDetected = digitalRead(WATER_PIN) == HIGH;

  // ================= READ KY-015 (TEMP/HUMID) =================
  static unsigned long lastDHTRead = 0;
  if(millis() - lastDHTRead >= 2000){ // DHT11 max once every 2s
    lastDHTRead = millis();
    float t = ky015.readTemperature();
    float h = ky015.readHumidity();
    if(!isnan(t)) temperature = t;
    if(!isnan(h)) humidity = h;
  }

  // ================= SERIAL MONITOR =================
  Serial.print("Distance: ");
  if(distance < 0) Serial.print("Out of range");
  else Serial.print(distance,1);
  Serial.print(" cm, Water: ");
  Serial.print(waterDetected ? "YES" : "NO");
  Serial.print(", Temp: "); Serial.print(temperature);
  Serial.print(" C, Humidity: "); Serial.println(humidity);

  // ================= OLED DISPLAY =================
  display.clearDisplay();
  display.setTextSize(1); // normal size for all text

  display.setCursor(0,0);
  display.print("Distance: ");
  if(distance < 0) display.println("---");
  else display.print(distance,1), display.println(" cm");

  display.setCursor(0,15);
  display.print("Water: ");
  display.println(waterDetected ? "YES" : "NO");

  display.setCursor(0,30);
  display.print("Temp: ");
  display.print(temperature,1);
  display.print("C");

  display.setCursor(0,45);
  display.print("Humidity: ");
  display.print(humidity,0);
  display.println("%");

  // WiFi status line (only addition to the display; delete if not wanted)
  display.setCursor(0,56);
  if (wm.getConfigPortalActive()) display.print("AP:" AP_NAME);
  else if (WiFi.status() == WL_CONNECTED) display.print("WiFi: connected");
  else display.print("WiFi: offline");

  display.display();

  // ================= FIREBASE UPDATE =================
  unsigned long currentMillis = millis();
  if(currentMillis - previousFirebase >= FIREBASE_INTERVAL){
    previousFirebase = currentMillis;
    if(WiFi.status() == WL_CONNECTED){
      HTTPClient http;
      http.begin(firebaseURL);
      http.addHeader("Content-Type","application/json");

      String jsonData = "{";
      jsonData += "\"distance\":" + String(distance,1) + ",";
      jsonData += "\"water\":" + String(waterDetected ? 1 : 0) + ",";
      jsonData += "\"temperature\":" + String(temperature,1) + ",";
      jsonData += "\"humidity\":" + String(humidity,0);
      jsonData += "}";

      int httpResponseCode = http.PUT(jsonData);
      Serial.print("Firebase Response: ");
      Serial.println(httpResponseCode);

      http.end();
    } else {
      Serial.println("WiFi Disconnected");
    }
  }

  delay(10); // very short delay to keep OLED instant
}
